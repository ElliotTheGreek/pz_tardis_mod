module TARDIS
{
    /*  Both shapes are world models rather than tile sprites. That keeps the
        exterior to a single 1x1 tile it can be summoned onto, and lets the
        console sit in the middle of the control room without needing a
        custom tileset packed with TileZed.

        The meshes live in media/models_X and the textures in media/textures;
        both are generated by tools/gen_model.py and tools/gen_console.py.

        modelScale is the one number to touch if the shell reads too large or
        too small in game: the meshes are authored at 1 unit == 1 tile.  */

    model TARDISPoliceBoxModel
    {
        mesh = TARDIS_PoliceBox,
        texture = TARDIS_PoliceBox,
        scale = 1.0,
    }

    model TARDISConsoleModel
    {
        mesh = TARDIS_Console,
        texture = TARDIS_Console,
        scale = 1.0,
    }

    item TARDISPoliceBox
    {
        ItemType = base:normal,
        Type = Normal,
        DisplayName = TARDIS,
        DisplayCategory = Furniture,
        Icon = TARDIS_PoliceBox,
        Weight = 60.0,
        WorldStaticModel = TARDISPoliceBoxModel,
        StaticModel = TARDISPoliceBoxModel,
        Tooltip = Tooltip_TARDIS_PoliceBox,
    }

    item TARDISConsole
    {
        ItemType = base:normal,
        Type = Normal,
        DisplayName = TARDIS Console,
        DisplayCategory = Furniture,
        Icon = TARDIS_PoliceBox,
        Weight = 80.0,
        WorldStaticModel = TARDISConsoleModel,
        StaticModel = TARDISConsoleModel,
    }
}
